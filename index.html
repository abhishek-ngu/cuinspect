using namespace llvm;
using namespace llvm::object;

int main(int argc, char** argv) {
    if (argc != 2) return 1;

    // Load file into memory buffer
    auto BufferOrErr = MemoryBuffer::getFile(argv[1]);
    if (BufferOrErr.getError()) return 1;

    // Parse as ELF object file
    auto ObjOrErr = ObjectFile::createELFObjectFile(BufferOrErr.get()->getMemBufferRef());
    if (!ObjOrErr) return 1;

    ELFObjectFileBase* ELF = dyn_cast<ELFObjectFileBase>(ObjOrErr.get().get());

    // Iterate all sections
    for (const SectionRef& Sec : ELF->sections()) {
        StringRef Name; Sec.getName(Name);
        uint64_t Size; Sec.getSize(Size);
        if (Size == 0) continue;

        std::cout << "  " << std::left << std::setw(20) << Name.str()
                  << std::right << std::setw(10) << Size << " bytes\n";

        // Special handling for GPU sections
        if (Name == ".nv.ptx") {
            StringRef Data; Sec.getContents(Data);
            std::cout << "    [PTX]: " << Data.substr(0, 200).str() << "...\n";
        }
    }

    // Extract DWARF debug info sections
    for (const SectionRef& Sec : ELF->sections()) {
        StringRef Name; Sec.getName(Name);
        if (Name.startswith(".debug_")) {
            uint64_t Size; Sec.getSize(Size);
            std::cout << "  DWARF: " << Name.str() << " " << Size << " bytes\n";
        }
    }
    return 0;
}

Build & Run

bash 
# 1. Install LLVM 18+ (Ubuntu/WSL)
sudo apt install llvm-18-dev clang-18

# 2. Clone & build
git clone https://github.com/YOUR_USERNAME/cuinspect
cd cuinspect
mkdir build && cd build
cmake .. -DLLVM_DIR=/usr/lib/llvm-18/lib/cmake/llvm
make -j$(nproc)

# 3. Create test kernel
cat > test.cu << 'EOF'
__global__ void add_kernel(float* a, float* b, float* c, int n) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i < n) c[i] = a[i] + b[i];
}
EOF

# 4. Compile to CUBIN (not just PTX!) with debug info
nvcc -cubin -arch=sm_90 -g -o test.cubin test.cu

# 5. Inspect!
./cuinspect test.cubin

CV-Ready Description

cuinspect C++ · LLVM · ELF/DWARF · CUDA
  • Built a tool parsing NVIDIA CUDA binaries (ELF format) to extract embedded PTX/SASS sections and DWARF debug info
  • Implements section enumeration, symbol extraction, and kernel metadata analysis (registers, shared memory, theoretical occupancy)
  • Uses LLVM's Object library for robust ELF/DWARF parsing; demonstrates understanding of GPU binary layout and NVPTX-specific DWARF extensions
  • Handles CUDA-specific sections: .nv.ptx, .nv.sass, .nv.info, .nv.reloc and DWARF ops like DW_OP_NVPTX_lane_id, DW_OP_NVPTX_warp_id
  • SASS instruction mix analysis via nvdisasm integration
  • JSON output mode for CI/CD pipeline integration
